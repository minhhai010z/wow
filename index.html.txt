  <script>
    function celebrate() {
      document.getElementById("wish").style.display = "block";

      for (let i = 0; i < 35; i++) {
        setTimeout(createBalloon, i * 100);
      }

      for (let i = 0; i < 6; i++) {
        setTimeout(firework, i * 350);
      }
    }

    function createBalloon() {
      const balloon = document.createElement("div");
      balloon.className = "balloon";
      balloon.textContent = ["🎈", "💗", "💖", "🎉"][Math.floor(Math.random()*4)];
      balloon.style.left = Math.random()*100 + "vw";
      balloon.style.animationDuration = (3 + Math.random()*4) + "s";
      document.body.appendChild(balloon);
      setTimeout(() => balloon.remove(), 7500);
    }

    function firework() {
      const x = Math.random()*window.innerWidth;
      const y = Math.random()*window.innerHeight*0.7;

      for (let i = 0; i < 35; i++) {
        const spark = document.createElement("div");
        spark.className = "spark";
        spark.style.left = x + "px";
        spark.style.top = y + "px";
        spark.style.background = `hsl(${Math.random()*360},100%,65%)`;

        const angle = Math.random()*Math.PI*2;
        const distance = 50 + Math.random()*180;
        spark.style.setProperty("--x", Math.cos(angle)*distance + "px");
        spark.style.setProperty("--y", Math.sin(angle)*distance + "px");

        document.body.appendChild(spark);
        setTimeout(() => spark.remove(), 1500);
      }
    }

    setInterval(createBalloon, 1000);
  </script>
